Core SQL Window CTE Script 

WITH CustomerOrders AS (
    SELECT customer_id, amount, order_date,
           LAG(order_date, 1) OVER (PARTITION BY customer_id ORDER BY order_date) AS prev_order_date
    FROM sales_transactions
)
SELECT customer_id, AVG(DATEDIFF(day, prev_order_date, order_date)) AS avg_repurchase_days,
       SUM(amount) AS total_clv
FROM CustomerOrders
GROUP BY customer_id HAVING AVG(DATEDIFF(day, prev_order_date, order_date)) > 60;
